<?php

session_start();

if(!isset($_SESSION['username'])){
    header("Location: login_admin.php");
    exit;
}


/* ==================================================
   KONEKSI DATABASE
================================================== */

include __DIR__ . "/koneksi.php";


/* ==================================================
   CEK ID
================================================== */

if(!isset($_GET['id']) || !is_numeric($_GET['id'])){

    header("Location: berita.php");

    exit;

}

$id = intval($_GET['id']);


/* ==================================================
   AMBIL DATA BERITA
================================================== */

$query = mysqli_query(
    $koneksi,
    "SELECT * FROM berita WHERE id=$id"
);

$data = mysqli_fetch_assoc($query);


if(!$data){

    header("Location: berita.php");

    exit;

}


$pesan = "";


/* ==================================================
   UPDATE BERITA
================================================== */

if(isset($_POST['update'])){


    /* ==============================
       DATA FORM
    ============================== */

    $judul = mysqli_real_escape_string(
        $koneksi,
        trim($_POST['judul'])
    );


    $isi = mysqli_real_escape_string(
        $koneksi,
        trim($_POST['isi'])
    );


    $tanggal = mysqli_real_escape_string(
        $koneksi,
        $_POST['tanggal']
    );


    $foto_lama = $data['foto'];

    $foto = $foto_lama;



    /* ==============================
       VALIDASI DATA
    ============================== */

    if($judul == ""){

        $pesan = "Judul berita tidak boleh kosong.";

    }

    elseif($isi == ""){

        $pesan = "Isi berita tidak boleh kosong.";

    }

    elseif($tanggal == ""){

        $pesan = "Tanggal berita harus diisi.";

    }



    /* ==============================
       UPLOAD FOTO BARU
    ============================== */

    if(
        $pesan == "" &&
        isset($_FILES['foto']) &&
        $_FILES['foto']['error'] != UPLOAD_ERR_NO_FILE
    ){

        $nama_file = $_FILES['foto']['name'];

        $tmp_file = $_FILES['foto']['tmp_name'];

        $ukuran = $_FILES['foto']['size'];


        $ext = strtolower(
            pathinfo(
                $nama_file,
                PATHINFO_EXTENSION
            )
        );


        $format = array(
            'jpg',
            'jpeg',
            'png',
            'webp'
        );


        /* VALIDASI FORMAT */

        if(!in_array($ext, $format)){

            $pesan =
                "Format foto tidak diperbolehkan. "
                . "Gunakan JPG, JPEG, PNG, atau WEBP.";

        }


        /* VALIDASI UKURAN */

        elseif($ukuran > 2 * 1024 * 1024){

            $pesan =
                "Ukuran foto maksimal 2 MB.";

        }


        /* VALIDASI UPLOAD */

        elseif(!is_uploaded_file($tmp_file)){

            $pesan =
                "Foto gagal diupload.";

        }


        else{


            /* NAMA FILE BARU */

            $foto_baru =
                time()
                . "_"
                . uniqid()
                . "."
                . $ext;


            $folder =
                __DIR__ . "/images/";


            /* PASTIKAN FOLDER ADA */

            if(!is_dir($folder)){

                mkdir(
                    $folder,
                    0755,
                    true
                );

            }


            /* PINDAHKAN FOTO */

            if(move_uploaded_file(
                $tmp_file,
                $folder . $foto_baru
            )){


                $foto = $foto_baru;


                /* ==============================
                   HAPUS FOTO LAMA
                ============================== */

                if(
                    !empty($foto_lama) &&
                    file_exists(
                        $folder . $foto_lama
                    )
                ){

                    unlink(
                        $folder . $foto_lama
                    );

                }

            }

            else{

                $pesan =
                    "Foto gagal disimpan.";

            }

        }

    }



    /* ==============================
       UPDATE DATABASE
    ============================== */

    if($pesan == ""){


        $sql = "

            UPDATE berita SET

                judul='$judul',

                isi='$isi',

                foto='$foto',

                tanggal='$tanggal'

            WHERE id=$id

        ";


        if(mysqli_query(
            $koneksi,
            $sql
        )){


            header(
                "Location: berita.php"
            );

            exit;


        }

        else{


            $pesan =
                "Gagal mengubah berita: "
                . mysqli_error($koneksi);

        }

    }

}
